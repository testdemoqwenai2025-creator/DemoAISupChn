   177→// ============== THEME ==============
   178→export type Theme = 'dark' | 'light'
   179→
   180→export const ThemeContext = React.createContext<{
   181→  theme: Theme
   182→  toggleTheme: () => void
   183→}>({ theme: 'dark', toggleTheme: () => {} })
   184→
   185→export function ThemeProvider({ children }: { children: React.ReactNode }) {
   186→  const [theme, setTheme] = useState<Theme>('dark')
   187→  return (
   188→    <ThemeContext.Provider value={{ theme, toggleTheme: () => setTheme(t => t === 'dark' ? 'light' : 'dark') }}>
   189→      {children}
   190→    </ThemeContext.Provider>
   191→  )
   192→}
   193→
   194→export function useTheme() {
   195→  return React.useContext(ThemeContext)
   196→}
   197→
   198→export function ThemeToggle() {
   199→  const { theme, toggleTheme } = useTheme()
   200→  return (
   201→    <Button variant="ghost" size="icon" onClick={toggleTheme}>
   202→      {theme === 'dark' ? <Sun className="w-5 h-5" /> : <Moon className="w-5 h-5" />}
   203→    </Button>
   204→  )
   205→}

   212→export function RiskBadge({ level }: { level: string }) {
   213→  const config = {
   214→    low: { color: 'bg-green-500/20 text-green-400 border-green-500/30', icon: CheckCircle2 },
   215→    medium: { color: 'bg-yellow-500/20 text-yellow-400 border-yellow-500/30', icon: AlertTriangle },
   216→    high: { color: 'bg-orange-500/20 text-orange-400 border-orange-500/30', icon: AlertTriangle },
   217→    critical: { color: 'bg-red-500/20 text-red-400 border-red-500/30', icon: XCircle }
   218→  }
   219→  const { color, icon: Icon } = config[level as keyof typeof config] || config.medium
   220→  return <Badge variant="outline" className={`${color} gap-1`}><Icon className="w-3 h-3" />{level.toUpperCase()}</Badge>
   221→}

   864→      </DialogContent>
   865→    </div>
   866→  )
